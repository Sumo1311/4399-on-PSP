cmake_minimum_required(VERSION 3.16)
project(AgeOfWar2PSP LANGUAGES C CXX)
file(GLOB_RECURSE PORT_SOURCES CONFIGURE_DEPENDS src/*.cpp src/*.c)
add_executable(psp_replica ${PORT_SOURCES})
target_compile_features(psp_replica PRIVATE cxx_std_17)
target_compile_options(psp_replica PRIVATE -Wall -Wextra -ffp-contract=off)
target_link_libraries(psp_replica PRIVATE pspgu pspge pspdisplay pspctrl psprtc psppower pspdebug pspaudio pspnet pspnet_adhoc pspnet_adhocctl pspwlan z lz4 psputility)
set(XMB_ARGS)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/ICON0.png")
  list(APPEND XMB_ARGS ICON_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/ICON0.png")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/PIC1.png")
  list(APPEND XMB_ARGS BACKGROUND_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/PIC1.png")
endif()
target_link_options(psp_replica PRIVATE "-Wl,--wrap=__libcglue_init" "-Wl,--wrap=_init" "-Wl,-Map=${CMAKE_BINARY_DIR}/age_of_war_2.map")
create_pbp_file(TARGET psp_replica TITLE "战争进化史2 内存修复10" VERSION 01.03 MEMSIZE 1 ${XMB_ARGS})
