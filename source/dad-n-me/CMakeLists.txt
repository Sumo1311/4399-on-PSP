cmake_minimum_required(VERSION 3.16)
project(DadNMePSP VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
option(DNM_FUNCTION_PROFILE "QA-only AVM1 function timing" OFF)
option(DNM_SMOOTH_VISUALS "Reduce expensive visual effects for PSP-2000 frame pacing" ON)
set(DNM_TEXTURE_CACHE_MB 24 CACHE STRING "Decoded texture LRU budget in MiB")
set(DNM_RESIDENT_PCM_MB 10 CACHE STRING "Short PCM preload budget in MiB")
if(DNM_TEXTURE_CACHE_MB LESS 4 OR DNM_TEXTURE_CACHE_MB GREATER 24 OR DNM_RESIDENT_PCM_MB LESS 0 OR DNM_RESIDENT_PCM_MB GREATER 10)
    message(FATAL_ERROR "Invalid fixed-heap texture or PCM budget")
endif()
set(DNM_SCENE_SLOTS 4 CACHE STRING "Bounded source/render scene slots")
if(DNM_SCENE_SLOTS LESS 2 OR DNM_SCENE_SLOTS GREATER 8)
    message(FATAL_ERROR "DNM_SCENE_SLOTS must be between 2 and 8")
endif()
set(DNM_SOURCE_LOOKAHEAD 1 CACHE STRING "Future source scenes after the opening cache warmup")
if(DNM_SOURCE_LOOKAHEAD LESS 1 OR NOT DNM_SOURCE_LOOKAHEAD LESS DNM_SCENE_SLOTS)
    message(FATAL_ERROR "DNM_SOURCE_LOOKAHEAD must be between 1 and DNM_SCENE_SLOTS-1")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(cmake/engine.cmake)
add_library(dnm_engine STATIC ${engine_sources} src/platform.cpp src/source_accelerators.cpp src/source_visual_policy.cpp)
if(DNM_FUNCTION_PROFILE)
    target_compile_definitions(dnm_engine PUBLIC DNM_FUNCTION_PROFILE=1)
endif()
target_include_directories(dnm_engine PUBLIC vendor/gameswf)
target_compile_definitions(dnm_engine PUBLIC
    TU_CONFIG_LINK_TO_THREAD=0 TU_USE_SDL=0 TU_CONFIG_LINK_STATIC=1
    TU_CONFIG_LINK_TO_FREETYPE=0 TU_CONFIG_LINK_TO_LIBPNG=0
    TU_ENABLE_NETWORK=0 TU_CONFIG_VERBOSE=1 GAMESWF_SWF6_NUMERIC=1)
target_compile_options(dnm_engine PRIVATE -w -fno-strict-aliasing -ffp-contract=off)
if(PSP)
    option(DNM_HEADLESS "Disable rendering to isolate runtime qualification" OFF)
    target_link_libraries(dnm_engine PUBLIC jpeg z m)
    add_executable(dad_n_me src/psp_main_threaded.cpp src/render_psp.cpp src/audio_psp.cpp src/device_font.cpp vendor/lz4/lz4.c)
    target_compile_definitions(dad_n_me PRIVATE $<$<BOOL:${DNM_HEADLESS}>:DNM_HEADLESS=1>)
    target_compile_definitions(dad_n_me PRIVATE DNM_TEXTURE_CACHE_MB=${DNM_TEXTURE_CACHE_MB} DNM_RESIDENT_PCM_MB=${DNM_RESIDENT_PCM_MB} DNM_SCENE_SLOTS=${DNM_SCENE_SLOTS} DNM_SOURCE_LOOKAHEAD=${DNM_SOURCE_LOOKAHEAD} DNM_SMOOTH_VISUALS=$<BOOL:${DNM_SMOOTH_VISUALS}>)
    target_compile_options(dad_n_me PRIVATE -Wall -Wextra -fno-strict-aliasing -ffp-contract=off)
    target_link_libraries(dad_n_me PRIVATE dnm_engine pspgu pspge pspdisplay pspctrl psppower pspdebug pspaudio)
    target_link_options(dad_n_me PRIVATE -Wl,-Map=${CMAKE_BINARY_DIR}/dad_n_me.map)
    create_pbp_file(TARGET dad_n_me TITLE "Dad n Me Response R24 Test" VERSION 01.35 MEMSIZE 1)
else()
    target_include_directories(dnm_engine PRIVATE /opt/homebrew/include)
    target_link_directories(dnm_engine PUBLIC /opt/homebrew/lib)
    target_link_libraries(dnm_engine PUBLIC jpeg z)
    add_executable(dnm_probe tests/probe.cpp)
    target_link_libraries(dnm_probe PRIVATE dnm_engine)
    add_executable(dnm_inspect tests/inspect.cpp)
    target_link_libraries(dnm_inspect PRIVATE dnm_engine)
    add_executable(dnm_lookup_test tests/lookup.cpp)
    target_link_libraries(dnm_lookup_test PRIVATE dnm_engine)
    add_executable(dnm_mask_dispatch_test tests/mask_dispatch.cpp)
    target_link_libraries(dnm_mask_dispatch_test PRIVATE dnm_engine)
    add_executable(dnm_liveness_test tests/liveness.cpp)
    target_link_libraries(dnm_liveness_test PRIVATE dnm_engine)
    add_executable(dnm_vm_strings_test tests/vm_strings.cpp)
    target_link_libraries(dnm_vm_strings_test PRIVATE dnm_engine)
    add_executable(dnm_bounds_cache_test tests/bounds_cache.cpp)
    target_link_libraries(dnm_bounds_cache_test PRIVATE dnm_engine)
    add_executable(dnm_frame_event_test tests/frame_event.cpp)
    target_link_libraries(dnm_frame_event_test PRIVATE dnm_engine)
endif()

if(NOT PSP)
    add_executable(dnm_advance_snapshot_test tests/advance_snapshot.cpp)
    target_link_libraries(dnm_advance_snapshot_test PRIVATE dnm_engine)
endif()

if(NOT PSP)
    add_executable(dnm_recorded_motion_test tests/recorded_motion.cpp)
    target_link_libraries(dnm_recorded_motion_test PRIVATE dnm_engine)
endif()
