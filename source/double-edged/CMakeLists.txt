cmake_minimum_required(VERSION 3.16)
project(DoubleEdgedPSP LANGUAGES C CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(DE_TEXTURE_CACHE_MB 22 CACHE STRING "Dedicated decoded texture arena in MiB")
set(DE_RESIDENT_PCM_MB 4 CACHE STRING "Maximum short PCM retained in RAM, in MiB")
if(DE_RESIDENT_PCM_MB LESS 0 OR DE_RESIDENT_PCM_MB GREATER 4)
    message(FATAL_ERROR "DE_RESIDENT_PCM_MB must be between 0 and 4")
endif()
set(DE_RESIDENT_BANK_MB 3 CACHE STRING "Maximum compressed art bank retained in RAM, in MiB")
if(DE_RESIDENT_BANK_MB LESS 0 OR DE_RESIDENT_BANK_MB GREATER 16)
    message(FATAL_ERROR "DE_RESIDENT_BANK_MB must be between 0 and 16")
endif()
set(DE_SCENE_SLOTS 4 CACHE STRING "Bounded source/render scene slots; one slot is currently displayed")
set(DE_GC_PHASE 0 CACHE STRING "Collection phase within the unchanged 30-source-tick period")
option(DE_REALTIME_FRONTEND "Original30Hz clock with one pending latest scene" ON)
option(DE_ADAPTIVE_GC "Bound forced full GC by tracked growth and120 source ticks" ON)
option(DE_LATE_CAPTURE "Defer at most one visual capture while source clock is late" ON)
option(DE_NATIVE_TIMELINE "Traverse original timeline enter order in native code" ON)
option(DE_BATCH_RENDER_SYNC "Call the trusted native mirror updater in one source flush" OFF)
option(DE_DIRECT_BAKED_CAPTURE "Record baked static shapes without a generic character wrapper" OFF)
set(DE_AUDIO_DELAY_BLOCKS 3 CACHE STRING "PCM transport delay blocks; voice semantics unchanged")
set(DE_BUILD_TITLE "Double Edged Response B13" CACHE STRING "XMB candidate title")
if(DE_GC_PHASE LESS 0 OR DE_GC_PHASE GREATER 29)
    message(FATAL_ERROR "DE_GC_PHASE must be between 0 and 29")
endif()
if(DE_SCENE_SLOTS LESS 4 OR DE_SCENE_SLOTS GREATER 8)
    message(FATAL_ERROR "DE_SCENE_SLOTS must be between 4 and 8")
endif()
option(DE_ENABLE_EFFECTS "Enable source group filters" ON)
option(DE_CONST_JSON "Retain immutable parsed source assets outside cyclic GC scans" OFF)
option(DE_CRISP_ART "Use nearest magnification for baked pixel art; retain glyph filtering" OFF)
option(DE_PREPARE_TEXTURES "Decode immutable queued-scene textures on the source thread before publication" OFF)
option(DE_COMPACT_TEXTURE_ARENA "Allocate texture/CLUT storage in256-byte ranges with adjacent free-range coalescence" OFF)
option(DE_ART_SHAPE_INDEX "Index immutable shape groups at startup for exact raster-entry lookup" OFF)
set(DE_RASTER_DENSITY_SCALE 1.0 CACHE STRING "Scale selected source-art raster density without changing geometry")
option(DE_WLAN "Build the two-PSP cooperative frontend" OFF)
set(QJS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/quickjs-ng")
add_library(source_js STATIC "${QJS_ROOT}/quickjs.c" "${QJS_ROOT}/dtoa.c" "${QJS_ROOT}/libregexp.c" "${QJS_ROOT}/libunicode.c")
target_include_directories(source_js PUBLIC "${QJS_ROOT}")
target_compile_definitions(source_js PRIVATE _GNU_SOURCE NDEBUG __STDC_NO_ATOMICS__=1)
target_compile_options(source_js PRIVATE -O3 -fno-strict-aliasing -ffp-contract=off)
option(DE_CPU_PROBE "Build an isolated source CPU/function diagnostic, never a game release" OFF)
if(DE_CPU_PROBE)
    add_executable(double_edged tools/quickjs_scene_probe.c)
    target_compile_definitions(double_edged PRIVATE DE_PROBE_CONST_JSON=1 DE_PROBE_TYPED_SLOTS=1 DE_PROBE_FUNCTION_PROFILE=1)
    target_compile_options(double_edged PRIVATE -O2 -ffp-contract=off)
    target_link_libraries(double_edged PRIVATE source_js pspdebug pspdisplay pspge psppower psprtc m)
    target_link_options(double_edged PRIVATE "-Wl,-Map=${CMAKE_BINARY_DIR}/double_edged.map")
    create_pbp_file(TARGET double_edged TITLE "Double Edged CPU DIAGNOSTIC" VERSION 00.01 MEMSIZE 1)
    return()
endif()
include(cmake/engine.cmake)
add_library(drawing_engine STATIC ${engine_sources} src/platform.cpp)
target_include_directories(drawing_engine PUBLIC vendor/gameswf)
target_compile_definitions(drawing_engine PUBLIC TU_CONFIG_LINK_TO_THREAD=0 TU_USE_SDL=0 TU_CONFIG_LINK_STATIC=1 TU_CONFIG_LINK_TO_FREETYPE=0 TU_CONFIG_LINK_TO_LIBPNG=0 TU_ENABLE_NETWORK=0 TU_CONFIG_VERBOSE=1 GAMESWF_SWF6_NUMERIC=0 GAMESWF_SWF8_CASE_SENSITIVE=1 GAMESWF_SHARED_AVM1_STACK=1)
target_compile_options(drawing_engine PRIVATE -w -fno-strict-aliasing -ffp-contract=off)
target_link_libraries(drawing_engine PUBLIC jpeg z m)
add_executable(double_edged src/psp_main_threaded.cpp src/source_vm.cpp src/storage.cpp src/audio_psp.cpp src/render_bridge.cpp src/render_psp.cpp src/device_font.cpp src/filter_pixels.cpp vendor/lz4/lz4.c)
target_include_directories(double_edged PRIVATE src)
target_compile_definitions(double_edged PRIVATE DE_SCENE_SLOTS=${DE_SCENE_SLOTS} DE_GC_PHASE=${DE_GC_PHASE} DE_CONST_JSON=$<BOOL:${DE_CONST_JSON}> DE_NATIVE_PAINT=1 DE_WLAN=$<BOOL:${DE_WLAN}> GLORTON_WLAN=$<BOOL:${DE_WLAN}> GLORTON_TEXTURE_CACHE_MB=${DE_TEXTURE_CACHE_MB} GLORTON_TEXTURE_POOL_MB=2 GLORTON_DEFER_TEXTURE_FREE=1)
target_compile_definitions(double_edged PRIVATE DE_REALTIME_FRONTEND=$<BOOL:${DE_REALTIME_FRONTEND}> DE_ADAPTIVE_GC=$<BOOL:${DE_ADAPTIVE_GC}> DE_LATE_CAPTURE=$<BOOL:${DE_LATE_CAPTURE}> DE_AUDIO_DELAY_BLOCKS=${DE_AUDIO_DELAY_BLOCKS})
target_compile_definitions(double_edged PRIVATE DE_NATIVE_TIMELINE=$<BOOL:${DE_NATIVE_TIMELINE}>)
target_compile_definitions(double_edged PRIVATE DE_BATCH_RENDER_SYNC=$<BOOL:${DE_BATCH_RENDER_SYNC}> DE_DIRECT_BAKED_CAPTURE=$<BOOL:${DE_DIRECT_BAKED_CAPTURE}>)
target_compile_definitions(double_edged PRIVATE DE_CRISP_ART=$<BOOL:${DE_CRISP_ART}>)
target_compile_definitions(double_edged PRIVATE DE_RASTER_DENSITY_SCALE=${DE_RASTER_DENSITY_SCALE})
target_compile_definitions(double_edged PRIVATE DE_PREPARE_TEXTURES=$<BOOL:${DE_PREPARE_TEXTURES}>)
target_compile_definitions(double_edged PRIVATE DE_COMPACT_TEXTURE_ARENA=$<BOOL:${DE_COMPACT_TEXTURE_ARENA}>)
target_compile_definitions(double_edged PRIVATE DE_ART_SHAPE_INDEX=$<BOOL:${DE_ART_SHAPE_INDEX}>)
target_compile_definitions(double_edged PRIVATE DE_RESIDENT_BANK_MB=${DE_RESIDENT_BANK_MB} DE_RESIDENT_PCM_MB=${DE_RESIDENT_PCM_MB})
if(DE_WLAN)
    target_sources(double_edged PRIVATE src/wlan_session.cpp src/wlan_frontend.cpp)
endif()
if(DE_ENABLE_EFFECTS)
    target_compile_definitions(double_edged PRIVATE DE_NATIVE_EFFECTS=1)
endif()
target_compile_options(double_edged PRIVATE -O3 -fno-strict-aliasing -ffp-contract=off)
target_link_libraries(double_edged PRIVATE source_js drawing_engine pspgu pspge pspdebug pspdisplay psppower pspctrl pspaudio m)
if(DE_WLAN)
    target_link_libraries(double_edged PRIVATE pspnet pspnet_adhoc pspnet_adhocctl pspwlan psputility)
endif()
target_link_options(double_edged PRIVATE "-Wl,-Map=${CMAKE_BINARY_DIR}/double_edged.map")
if(DE_WLAN)
    create_pbp_file(TARGET double_edged TITLE "Double Edged Two PSP" VERSION 01.00 MEMSIZE 1)
else()
    create_pbp_file(TARGET double_edged TITLE "${DE_BUILD_TITLE}" VERSION 01.01 MEMSIZE 1)
endif()
